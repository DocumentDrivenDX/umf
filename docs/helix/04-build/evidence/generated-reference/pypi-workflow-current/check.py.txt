import hashlib, json, zipfile
from pathlib import Path
wheels = list(Path('python/dist').glob('*.whl'))
assert len(wheels) == 1, 'Expected one official core wheel'
sources = {'umf/' + p.name: p for p in Path('python/src/umf').iterdir() if p.is_file()}
for name in ['schema.json', 'semantic-type-reference.schema.json', 'extension-package.schema.json']:
    p = Path('spec/core') / name
    sources['umf/' + str(p)] = p
sources.update({'umf/' + str(p): p for p in Path('spec/metamodel/generated').glob('*.json')})
for extension in ['domain-pack', 'dataset-source']:
    p = Path('spec/extensions') / extension / 'schema.json'
    sources['umf/' + str(p)] = p
with zipfile.ZipFile(wheels[0]) as wheel:
    assert {n for n in wheel.namelist() if n.startswith('umf/')} == set(sources)
    for name, source in sources.items():
        assert wheel.read(name) == source.read_bytes(), name
print(json.dumps({'wheel': wheels[0].name, 'sha256': hashlib.sha256(wheels[0].read_bytes()).hexdigest(), 'exactSourceFiles': len(sources)}))
