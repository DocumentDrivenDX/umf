results = []
def observe(name, call):
    try:
        value = call()
        result = {'probe': name, 'result': repr(value)}
    except Exception as e:
        result = {'probe': name, 'exception': type(e).__name__, 'message': str(e)}
    results.append(result)
    return result

cls = next(x for x in tree.body if isinstance(x, ast.ClassDef) and x.name == 'LocalPostgresSupervisor')
methods = {x.name:x for x in cls.body if isinstance(x, ast.FunctionDef)}
def assigned(method, name, values):
    node = next(x for x in ast.walk(methods[method]) if isinstance(x, ast.Assign)
                and any(isinstance(t, ast.Name) and t.id == name for t in x.targets))
    return eval(compile(ast.Expression(node.value), '<exact-source-expression>', 'eval'), vars(n), values)

    sup=authority._native_supervisor
    sup._native_effect_bindings[id(reservation)]=binding
    outcome=sup._record_native_effect_outcome(reservation,'confirmed',{'profile':'sol-source-receipt/0.1'})
    state=authority.observe_original_request(sup.authority_context,reservation.effect_key)
    bridge_results.append({'terminalStatus':state.effect_status,
        'actualReceiptDigest':state.effect_receipt_digest,
        'nativeReceiptDigest':outcome.receipt_digest,
        'adapterRecoveryExpectedDigest':n._digest(outcome.receipt_digest)})

        'approvalDigest':approval.approval_digest})
