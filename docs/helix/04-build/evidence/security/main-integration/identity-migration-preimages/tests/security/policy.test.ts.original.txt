test('display labels cannot resolve a different document and revisions must match',()=>{
  // @covers US-055-AC2
  for(const mutation of ['document','revision','duplicate-document','missing-document']){
    const {policy,resolution}=securityFixture();
    if(mutation==='document')policy.rules[0]!.target[0]!.documentId='other';
    if(mutation==='revision')resolution.documents[0]!.revision='schema-2';
    if(mutation==='duplicate-document')resolution.documents.push(resolution.documents[0]!);
    if(mutation==='missing-document')resolution.documents=[];
    expect(inspectSecurityPolicy(policy,resolution).complete,mutation).toBe(false);
    expect(()=>requireSecurityInterpretation(policy,resolution),mutation).toThrow();
  }
});

test('unknown security meaning and opaque native archives survive JSON/YAML',()=>{
  // @covers US-055-AC9
  const {policy,resolution}=securityFixture();
  const source={...policy,future:{a:[null,'retained']},native:{text:'CREATE POLICY p;\n-- retained\n',dialect:'unknown'}};
  for(const format of ['json','yaml'] as const){
    const restored=readSecuritySource(writeSecuritySource(source,format),format);
    expect(restored).toEqual(copyJson(source));
    const result=inspectSecurityPolicy(restored,resolution);
    expect(result.valid).toBe(true);expect(result.complete).toBe(false);
    expect(result.source).toEqual(copyJson(source));expect(()=>requireSecurityInterpretation(restored,resolution)).toThrow();
  }
  const unfamiliar={...policy,version:'9.0.0',rules:[{futureOperator:'opaque'}]};
  expect(readSecuritySource(writeSecuritySource(unfamiliar))).toEqual(unfamiliar);
  expect(inspectSecurityPolicy(unfamiliar,resolution).valid).toBe(false);
});

test('unknown ontology/facet/key qualifiers refuse while original source remains retained',()=>{
  // @covers US-055-AC3
  for(const where of ['ontology','field','facet','key']){
    const {policy,resolution}=securityFixture();
    if(where==='ontology')(resolution.ontology as any).future=true;
    if(where==='field')(resolution.ontology.entities[0]!.fields[0] as any).future=true;
    if(where==='facet')resolution.documents[0]!.document.modules[0]!.elements.find(e=>e.id==='salary')!.facets={future:true};
    if(where==='key')(resolution.documents[0]!.document.modules[0]!.elements[0]!.keys as any)[0].future=true;
    const result=inspectSecurityPolicy(policy,resolution);expect(result.complete,where).toBe(false);
    expect(result.resolution).toEqual(copyJson(resolution));expect(()=>requireSecurityInterpretation(policy,resolution)).toThrow();
  }
});

test('registered policy preserves extension but core inspection requires explicit closure',()=>{
  // @covers US-055-AC9
  const {policy,resolution}=securityFixture();const doc=resolution.documents[0]!.document;
  doc.vocabularies['umf.security']={version:'0.1.0'};doc.extensions={'umf.security':policy as any};
  const result=validateDocument(doc,securityRegistry());
  expect(result.valid).toBe(true);expect(result.complete).toBe(false);
  expect(result.diagnostics.some(d=>d.code==='SECURITY_RESOLUTION_REQUIRED')).toBe(true);
});
