const canonical = (value:unknown):unknown => Array.isArray(value)
  ? value.map(canonical)
  : value !== null && typeof value === 'object'

function register(spec:RegistrySpec|undefined):Registry {
  const registry=new Registry();
  if(spec){
    const manifest:ExtensionPackage={id:spec.id,version:spec.version,coreVersion:'0.1.0',description:'Frozen cross-runtime baseline callback',schema:true,semantics:'Baseline callback that exposes exact source context',scopes:[spec.scope],capabilities:{validation:'semantic',directions:[],evidence:['metamodel-authority-baseline']}};
    registry.register(manifest, (payload,context)=>[{code:'BASELINE_CALLBACK',path:context.path,message:`${context.document.id}:${context.scope}:${(payload as Record<string,Json>).kind}`,severity:'warning'}]);
  }
  return registry;
}

function tsResults() {
  return cases.map(row=>{
    const validation=validateDocument(row.document,register(row.registry));
    const result:Record<string,unknown>={id:row.id,valid:validation.valid,complete:validation.complete,diagnostics:validation.diagnostics};
    if(row.uuidLiteralProbe){
      const element=row.document.modules.find(m=>m.id===row.uuidLiteralProbe!.module)!.elements.find(e=>e.id===row.uuidLiteralProbe!.element)!;
      const evaluate=(doc:Document, target:typeof element)=>{try{checkSchemaLiteral(doc,target,row.uuidLiteralProbe!.value);return {accepted:true};}catch(error){return {accepted:false,error:error instanceof Error?error.message:String(error)};}};
      const inherited=copyJson(row.document) as unknown as Document;
      inherited.umf='0.9.0';
      for(const module of inherited.modules)for(const field of module.elements)delete field.semanticTypes;
      const inheritedElement=inherited.modules.find(m=>m.id===row.uuidLiteralProbe!.module)!.elements.find(e=>e.id===row.uuidLiteralProbe!.element)!;
      result.uuidLiteralEntryPoints={direct010:evaluate(row.document,element),privateInheritedView:evaluate(inherited, inheritedElement),documentValidation:{valid:validation.valid,diagnostics:validation.diagnostics}};
    }
    if(row.defaultProbe){
      try { const r=resolveCoreDefault(row.document,row.defaultProbe.field,row.defaultProbe.state);result.defaultResolution={applied:r.applied,result:r.result}; }
      catch(error){result.defaultResolution={error:error instanceof Error?error.message:String(error)};}
    }
    return result;
  });
}

function baselineHashes() {
  const result:Record<string,{gitBlob:string;sha256:string}>={};
  for(const path of sourcePaths){
    const current=readFileSync(join(root,path));
    const committed=execFileSync('git',['show',`${expectedHead}:${path}`],{cwd:root});
    if(!current.equals(committed))throw new Error(`Baseline source is modified from ${expectedHead}: ${path}`);
    result[path]={gitBlob:execFileSync('git',['rev-parse',`${expectedHead}:${path}`],{cwd:root,encoding:'utf8'}).trim(),sha256:sha256(current)};
  }
  const scriptPath=relative(root,fileURLToPath(import.meta.url));
  result[scriptPath]={gitBlob:'untracked-capture-script',sha256:sha256(readFileSync(fileURLToPath(import.meta.url)))};
  return result;
}

function main() {
  const args=process.argv.slice(2);
  if(args.length!==2||args[0]!=='--output')throw new Error('Usage: bun scripts/metamodel/capture-baseline.ts --output <evidence-directory>');
  const requested=resolve(root,args[1]!);
  const relativeOutput=relative(root,requested);
  if(isAbsolute(relativeOutput)||!relativeOutput.startsWith(outputPrefix)||relativeOutput!==`${outputPrefix}baseline`)throw new Error(`Output must be the frozen baseline path under ${outputPrefix}`);
  const head=execFileSync('git',['rev-parse','HEAD'],{cwd:root,encoding:'utf8'}).trim();
  if(head!==expectedHead)throw new Error(`Expected source HEAD ${expectedHead}, got ${head}`);
  if(new Set(cases.map(row=>row.id)).size!==cases.length)throw new Error('Duplicate baseline case ID');
  const hashes=baselineHashes();
  const parent=dirname(requested);mkdirSync(parent,{recursive:true});
  const stage=mkdtempSync(join(parent,'.baseline.capture-'));
  try {
    const caseData={format:'umf.current-core-baseline/1',sourceHead:head,cases};
    writeFileSync(join(stage,'cases.json'),json(caseData),'utf8');
    const tsData={runtime:`Bun ${Bun.version}`,command,results:tsResults()};
    writeFileSync(join(stage,'ts-results.json'),json(tsData),'utf8');
    const pythonCasePath=join(stage,'cases.json'), pythonExecutable=locatePython();
    const python=spawnSync(pythonExecutable,['-c',pythonRunner,pythonCasePath],{cwd:root,encoding:'utf8',env:{...process.env,PYTHONPATH:join(root,'python/src'),UMF_PYTHON_SRC:join(root,'python/src')}});
    let pythonRows:unknown[]=[];
    if(python.status===0){try{pythonRows=JSON.parse(python.stdout);}catch(error){throw new Error(`Python output is not JSON: ${String(error)}\n${python.stdout}\n${python.stderr}`);}}
    else throw new Error(`Python baseline runner failed (${python.status}):\n${python.stdout}\n${python.stderr}`);
    const pythonData={runtime:pythonVersion(pythonExecutable),pythonExecutable,command:[pythonExecutable,'-c','<embedded frozen runner>','cases.json'],knownGaps:{unsupportedCoreProfiles:['0.9.0','0.10.0'],semanticSurfaces:['facets','typed literals/defaults','keys','relationships'],note:'These historical Python coverage gaps are recorded as observed outcomes, not parity targets.'},results:pythonRows};
    writeFileSync(join(stage,'python-results.json'),json(pythonData),'utf8');
    writeFileSync(join(stage,'source-hashes.json'),json({sourceHead:head,sources:hashes}),'utf8');
    const outputs=['cases.json','ts-results.json','python-results.json','source-hashes.json'];
    const outputHashes=Object.fromEntries(outputs.map(name=>[name,sha256(readFileSync(join(stage,name)))]));
    const receipt={id:'umf.metamodel-authority.baseline',status:'frozen-current-behavior-baseline',sourceHead:head,command,createdAt:new Date().toISOString(),caseCount:cases.length,typescriptResultCount:tsData.results.length,pythonResultCount:pythonRows.length,pythonExecutable,pythonExitCode:python.status,pythonStdout:python.stdout,pythonStderr:python.stderr,sourceHashes:hashes,outputHashes,uuidLiteralCorrectionReview:{caseId:'domains/uuid-literal-version-boundary',observed:tsData.results.find(row=>row.id==='domains/uuid-literal-version-boundary'),note:'Record both the direct 0.10.0 schema-literal entry point and the private inherited-profile view. A later correction must name the entry point, intended 0.10.0 UUID meaning, and reason; do not rewrite this receipt or its outputs.'}};
    writeFileSync(join(stage,'receipt.json'),json(receipt),'utf8');
    if(existsSync(requested))throw new Error(`Refusing to overwrite frozen baseline: ${relativeOutput}`);
    renameSync(stage,requested);
    process.stdout.write(json({status:'captured',output:relativeOutput,caseCount:cases.length,sourceCount:Object.keys(hashes).length,head}));
  } catch(error) {
    rmSync(stage,{recursive:true,force:true});
    throw error;
  }
}

function pythonVersion(executable:string):string {
  const result=spawnSync(executable,['--version'],{encoding:'utf8'});
  if(result.status!==0)throw new Error(`Cannot read Python version: ${result.stderr}`);
  return (result.stdout||result.stderr).trim();
}

function locatePython():string {
  const candidates:string[]=[];
  if(process.env.UMF_BASELINE_PYTHON)candidates.push(process.env.UMF_BASELINE_PYTHON);
  candidates.push('python3');
  for(const path of ['/private/tmp','/Users/erik/.cache/uv/archive-v0']){
    if(!existsSync(path))continue;
    for(const entry of readdirSync(path,{withFileTypes:true}))if(entry.isDirectory()){
      const candidate=join(path,entry.name,'bin','python');if(existsSync(candidate))candidates.push(candidate);
    }
  }
  for(const candidate of candidates){
    const result=spawnSync(candidate,['-c','import pydantic, jsonschema'],{cwd:root,encoding:'utf8'});
    if(result.status===0)return candidate;
  }
  throw new Error('No Python interpreter with the pinned pydantic/jsonschema runtime is available; set UMF_BASELINE_PYTHON to the project environment.');
}
