import {executeConstraints} from '/Users/erik/.codex/worktrees/d8be/umf/src/validation/constraint-kernel';
const root='/Users/erik/.codex/worktrees/d8be/umf';
const data=await Bun.file('/private/tmp/umf-key-relationship-program.json').json(),policy=await Bun.file('/private/tmp/umf-domains.json').json();
const V=(name:string)=>({var:name}),O=(op:string,...args:any[])=>({op,args}),G=(object:any,...path:any[])=>O('get',object,path),L=(name:string,value:any)=>({op:'let',name,value});
const elements=O('flatten',{op:'map',args:[G(V('doc'),'modules')],as:'module',index:'mi',value:{op:'map',args:[G(V('module'),'elements')],as:'node',index:'ei',value:{node:V('node'),module:V('module'),path:O('concat','/modules/',V('mi'),'/elements/',V('ei'))}}});
const index={op:'index',args:[V('elements')],as:'entry',value:O('tuple',G(V('entry'),'module','id'),G(V('entry'),'node','id'))};
const program={profile:'umf.constraints/1',steps:[L('elements',elements),L('elementIndex',index),...data.keySteps,...data.relationshipSteps]};
const rows=(await Bun.file(root+'/docs/helix/04-build/evidence/metamodel-authority/baseline-current/cases.json').json()).cases;
const expected=(await Bun.file(root+'/docs/helix/04-build/evidence/metamodel-authority/baseline-current/ts-results.json').json()).results;
const phase=(row:any)=>/^(?:KEY_|UNKNOWN_KEY_|RELATIONSHIP_|UNKNOWN_RELATIONSHIP_)/.test(row.code);
const stable=(v:any):any=>Array.isArray(v)?v.map(stable):v&&typeof v==='object'?Object.fromEntries(Object.keys(v).sort().map(k=>[k,stable(v[k])])):v;
const differences:any[]=[],skipped:string[]=[];
for(let i=0;i<rows.length;i++){
 if(expected[i].diagnostics.some((d:any)=>d.code.includes('STRUCTURE'))){skipped.push(rows[i].id);continue;}
 const actual=executeConstraints(rows[i].document,program,policy),want=expected[i].diagnostics.filter(phase);
 if(JSON.stringify(stable(actual))!==JSON.stringify(stable(want)))differences.push({id:rows[i].id,actual,want});
}
await Bun.write('/private/tmp/umf-key-program-parity.json',JSON.stringify({cases:rows.length,skippedStructural:skipped,compared:rows.length-skipped.length,differences},null,2));
console.log(JSON.stringify({cases:rows.length,compared:rows.length-skipped.length,differences:differences.length}));
